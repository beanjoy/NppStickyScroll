cmake_minimum_required(VERSION 3.21)
project(NppStickyScroll VERSION 0.4.1 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "NppStickyScroll requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(src/Version.h.in generated/Version.h @ONLY)
add_library(NppStickyScroll SHARED src/Plugin.cpp src/StickyView.cpp src/SettingsDialog.cpp src/SettingsDialog.rc src/Version.rc)
target_include_directories(NppStickyScroll PRIVATE src vendor/npp vendor/scintilla vendor/lexilla "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(NppStickyScroll PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(NppStickyScroll PRIVATE comctl32 gdi32 user32)
if(MSVC)
  target_compile_options(NppStickyScroll PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(NppStickyScroll PRIVATE /DYNAMICBASE /NXCOMPAT)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(FoldIndexTests tests/FoldIndexTests.cpp)
  target_include_directories(FoldIndexTests PRIVATE src vendor/scintilla)
  add_test(NAME FoldIndex COMMAND FoldIndexTests)
endif()
option(STICKY_HOST_TESTS "Run destructive tests ONLY in an isolated portable Notepad++" OFF)
if(STICKY_HOST_TESTS)
  target_sources(NppStickyScroll PRIVATE tests/HostTests.cpp)
  target_compile_definitions(NppStickyScroll PRIVATE STICKY_HOST_TESTS)
  target_link_libraries(NppStickyScroll PRIVATE version)
endif()
install(TARGETS NppStickyScroll RUNTIME DESTINATION NppStickyScroll)
